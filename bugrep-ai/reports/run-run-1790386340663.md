## 4. Root Cause Analysis

**Defective file:** [`src/cart.fixture.js`](src/cart.fixture.js), lines 13–16.

```js
function calculateTotal(items, discountPercentage) {
  let subtotal = items.reduce((sum, item) => sum + item.price, 0);
  let discount = (subtotal * discountPercentage) / 100;
  return subtotal - discount;
}
```

### Specific violations

| Defect | Lines | Business rule violated |
|---|---|---|
| No clamp: `subtotal - discount` can be negative when `discountPercentage > 100` | 15–16 | **Rule 2**: Cart totals must NEVER drop below 0 |
| `discountPercentage` is used without guarding for `undefined` / `null` → produces `NaN` | 15 | **Rule 4**: Missing discount treated as 0% |
| Negative `discountPercentage` passes through as a surcharge | 15–16 | **Rule 5**: Negative discount clamped to 0% |
| `NaN` / `±Infinity` discount is silently accepted | 15 | **Rule 6**: `NaN` or `±Infinity` must throw `RangeError` |
| Item prices are never validated — `undefined`, missing, or negative prices silently produce wrong results | 14 | **Rules 7 & 8**: Invalid/negative prices must throw `TypeError` |
| No rounding: raw floating-point arithmetic is returned | 16 | **Rule 9**: All totals rounded to 2 decimal places |

The primary reported failure (`-42.49`) traces directly to the missing clamp: `subtotal = 84.98`, `discount = 84.98 × 150 / 100 = 127.47`, `result = 84.98 − 127.47 = −42.49`.

 function calculateTotal(items, discountPercentage) {
-  let subtotal = items.reduce((sum, item) => sum + item.price, 0);
-  let discount = (subtotal * discountPercentage) / 100;
-  return subtotal - discount;
+  // Rule 6: NaN or ±Infinity discount → RangeError
+  if (typeof discountPercentage === 'number' &&
+      (isNaN(discountPercentage) || !isFinite(discountPercentage))) {
+    throw new RangeError('discountPercentage must be a finite number');
+  }
+
+  // Rule 4: undefined/missing discount → treat as 0%
+  // Rule 5: negative discount → clamp to 0%
+  let discount = (discountPercentage === undefined || discountPercentage === null)
+    ? 0
+    : discountPercentage;
+  if (discount < 0) discount = 0;
+
+  // Rule 3: empty cart always returns 0
+  if (items.length === 0) return 0;
+
+  // Rules 7 & 8: validate each item's price
+  for (const item of items) {
+    if (item.price === undefined || item.price === null || typeof item.price !== 'number' || isNaN(item.price)) {
+      throw new TypeError('Each item must have a numeric price');
+    }
+    if (item.price < 0) {
+      throw new TypeError('Item prices must be non-negative');
+    }
+  }
+
+  const subtotal = items.reduce((sum, item) => sum + item.price, 0);
+  const discountAmount = (subtotal * discount) / 100;
+  const total = subtotal - discountAmount;
+
+  // Rule 2: total must never be negative (discount > 100%)
+  const clamped = Math.max(0, total);
+
+  // Rule 9: round to 2 decimal places
+  return Math.round(clamped * 100) / 100;
 }
